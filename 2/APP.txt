                    }, HEARTBEAT_INTERVAL);
                }
                return () => clearInterval(interval);
            }, [isHost, step, roomName, maxSlots]);

            const connectToLobby = () => {
                if (lobbyPeerRef.current) return;
                const tempPeer = new Peer(null);
                tempPeer.on('open', (id) => {
                    const conn = tempPeer.connect(LOBBY_ID);
                    conn.on('open', () => {
                        conn.on('data', (data) => {
                            if (data.type === 'room_list') setActiveRooms(data.payload);
                        });
                        conn.send({ type: 'get_rooms' });
                    });
                    conn.on('error', () => becomeLobbyHost());
                });
                tempPeer.on('error', (err) => { 
                    if (err.type === 'peer-unavailable') becomeLobbyHost(); 
                });
                lobbyPeerRef.current = tempPeer;
            };

            const becomeLobbyHost = () => {
                if (lobbyPeerRef.current) lobbyPeerRef.current.destroy();
                const hostPeer = new Peer(LOBBY_ID);
                const currentRooms = new Map();
                hostPeer.on('open', () => {
                    setInterval(() => {
                        const now = Date.now();
                        let changed = false;
                        for (const [key, room] of currentRooms) {
                            if (now - room.lastPing > (HEARTBEAT_INTERVAL * 2)) {
                                currentRooms.delete(key);
                                changed = true;
                            }
                        }
                        if (changed) broadcastLobby(Array.from(currentRooms.values()));
                    }, 10000);
                });
                hostPeer.on('connection', (conn) => {
                    conn.on('data', (data) => {
                        if (data.type === 'announce_room') {
                            const room = { ...data.payload, lastPing: Date.now() };
                            currentRooms.set(data.payload.name, room);
                            broadcastLobby(Array.from(currentRooms.values()));
                        }
                        if (data.type === 'get_rooms') conn.send({ type: 'room_list', payload: Array.from(currentRooms.values()) });
                    });
                });
                const broadcastLobby = (list) => {
                    setActiveRooms(list);
                    for (const connId in hostPeer.connections) {
                        hostPeer.connections[connId].forEach(conn => { if(conn.open) conn.send({ type: 'room_list', payload: list }); });
                    }
                };
                lobbyPeerRef.current = hostPeer;

                            const displayStream = await navigator.mediaDevices.getDisplayMedia({ video: true, audio: true });
                            
                            // Guardamos el stream original para que no se muera

                            // Verificar si obtuvimos audio
                            if (displayStream.getAudioTracks().length === 0) {
                                displayStream.getTracks().forEach(t => t.stop()); // Limpieza
                                throw new Error("No se compartió audio. Asegúrate de marcar 'Compartir audio'.");
                            }

                            // TRUCO: Creamos un stream NUEVO solo con el audio.
                            // NO paramos el video track del original, porque Chrome mata el audio si lo haces.
                            // Al enviar solo 'newStream' a PeerJS, el video nunca se transmite. Privacidad 100%.

                            originalDisplayStreamRef.current = displayStream; // Mantener vivo

                            if (displayStream.getAudioTracks().length > 0) {
                                const sysSource = ctx.createMediaStreamSource(displayStream);
                                sysSource.connect(dest);
                            } else {
                                alert("Advertencia: No compartiste audio del sistema.");
                            }

                            // Mic

                            // Ojo: Si paramos el micStream, perdemos el input. No lo guardamos en ref global pero lo usa el contexto.
                            const micSource = ctx.createMediaStreamSource(micStream);
                            micSource.connect(dest);

                            newStream = dest.stream;
                            addLog("🔒 Modo Mezcla: Video bloqueado. Audio PC + Mic activos.", "success");

                        } catch (err) {
                            console.error(err);
                            alert("Error configurando mezcla: " + err.message);

                        }

                    addLogAndBroadcast(`${username}: ${logMsg}`, 'info');

                } catch (e) {
                    console.error("Error cambiando audio:", e);
                    alert("Error: " + e.message);
                    setAudioSourceType('none');
                }
            };

            // --- ACCIONES DE LOG ---
            const addLog = (text, type = 'info', id = null) => {
                const logId = id || Date.now() + Math.random(); // Generar ID si no existe
                
                // Evitar duplicados
                if (processedLogIds.current.has(logId)) return;
                processedLogIds.current.add(logId);
                
                // Limpiar caché de IDs viejos para no fugar memoria
                if (processedLogIds.current.size > 100) {
                    const arr = Array.from(processedLogIds.current);
                    processedLogIds.current = new Set(arr.slice(arr.length - 50));
                }

                setActivityLog(prev => [{ id: logId, time: new Date().toLocaleTimeString([], {hour:'2-digit', minute:'2-digit'}), text, type }, ...prev]);
            };

            const addLogAndBroadcast = (text, type = 'info') => {
                const id = Date.now() + Math.random();
                addLog(text, type, id); // Local
                // Enviar a otros
                const payload = { id, text, type };
                if (isHost) broadcast(payload, 'log_entry');
                else connectionsRef.current[0]?.send({ type: 'log_entry', payload });
            };

            // --- ACCIONES DE SALA ---
            const kickUser = (targetId) => {
                if (!isHost) return;
                const targetUser = usersInRoom.find(u => u.id === targetId);
                const conn = connectionsRef.current.find(c => c.peer === targetId);
                if (conn) {
                    conn.send({ type: 'kicked' });
                    setTimeout(() => conn.close(), 500); // Dar tiempo a recibir mensaje
                }
                handleDisconnect(targetId);
                addLogAndBroadcast(`🚫 ${targetUser?.name} fue expulsado`, 'danger');
            };

            const updateUserStatus = (uid, changes) => setUsersInRoom(prev => prev.map(u => u.id === uid ? { ...u, ...changes } : u));
            const broadcast = (payload, type) => connectionsRef.current.forEach(c => c.open && c.send({ type, payload }));

            const sendChat = () => {
                if (!newMessage.trim()) return;
                const msg = { id: Date.now(), sender: username, text: newMessage, time: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }) };
                setChatMessages(prev => [...prev, msg]);
                setNewMessage('');
                if (isHost) broadcast(msg, 'chat_message');
                else connectionsRef.current[0]?.send({ type: 'chat_message', payload: msg });
            };

            // --- SETUP ---
            const createRoom = async () => {
                if (!roomName.trim()) return;
                const peer = new Peer(roomName.trim());
                
                peer.on('open', (id) => {
                    setIsHost(true);
                    setMyPeerId(id);
                    // El Host empieza Muteado por defecto para evitar sustos
                    setUsersInRoom([{ id, name: username, isHost: true, isMuted: true }]);
                    setSpeakers([id]);
                    setAudioSourceType('none');
                    setStep('room');
                    addLog("Sala creada. Configura tu audio.", "success");
                });
                
                peer.on('connection', (conn) => {
                    // Evitar conexiones duplicadas si ya existe
                    if (connectionsRef.current.some(c => c.peer === conn.peer)) return;
                    
                    connectionsRef.current.push(conn);
                    conn.on('data', (data) => handleHostData(data, conn));
                    conn.on('close', () => handleDisconnect(conn.peer));
                    conn.on('error', () => handleDisconnect(conn.peer));
                });
                
                // Escuchar llamadas entrantes (Host escucha a Speakers)
                peer.on('call', call => {
                    call.answer(); 
                    call.on('stream', remoteStream => addAudioElement(call.peer, remoteStream));
                });

                peerRef.current = peer;
            };

            const joinRoom = async (targetRoom) => {
                if (!targetRoom) return;
                const peer = new Peer();
                peer.on('open', (id) => {
                    setMyPeerId(id);
                    setIsHost(false);
                    const conn = peer.connect(targetRoom);
                    
                    conn.on('open', () => {
                        setStep('room');
                        setRoomName(targetRoom);
                        conn.send({ type: 'join', name: username });
                        connectionsRef.current = [conn];
                        // Heartbeat para detectar desconexión del host
                        setInterval(() => {
                            if(!conn.open) { alert("Host desconectado"); cleanup(); }
                        }, 5000);
                    });
                    
                    conn.on('data', handleGuestData);
                    conn.on('close', () => { alert("Desconectado de la sala"); cleanup(); });
                    conn.on('error', () => { alert("Error de conexión"); cleanup(); });
                });
                
                peer.on('call', call => {
                    call.answer();
                    call.on('stream', remoteStream => addAudioElement(call.peer, remoteStream));
                });
                
                peerRef.current = peer;
            };

            // HANDLERS
            const handleHostData = (data, conn) => {
                const { type, payload } = data;
                
                // JOIN: Verificar duplicados estricto
                if (type === 'join') {
                    // Si el usuario ya está en la lista, ignorar (fix re-joins rápidos)
                    setUsersInRoom(prev => {
                         if (prev.some(u => u.id === conn.peer)) return prev;
                         
                         const newUser = { id: conn.peer, name: data.name, isHost: false, isMuted: true };
                         const updated = [...prev, newUser];
                         
                         // Broadcast del nuevo estado completo
                         broadcast(updated, 'users_update');
                         conn.send({ type: 'sync_log', payload: activityLog }); 
                         
                         // Log solo una vez aquí
                         addLogAndBroadcast(`${data.name} entró a la sala`, 'info');
                         return updated;
                    });
                }
                else if (type === 'chat_message') {
                    setChatMessages(prev => [...prev, payload]);
                    broadcast(payload, 'chat_message');
                }
                else if (type === 'mic_status') {
                    updateUserStatus(payload.userId, { isMuted: payload.isMuted });
                    broadcast(payload, 'mic_status');
                }
                else if (type === 'log_entry') {
                    addLog(payload.text, payload.type, payload.id);
                    broadcast(payload, 'log_entry');
                }
                else if (type === 'request_speak') {
                     updateUserStatus(conn.peer, { handRaised: true });
                     broadcast({ userId: conn.peer, handRaised: true }, 'hand_update');
                     addLogAndBroadcast(`${usersInRoom.find(u=>u.id===conn.peer)?.name} levantó la mano`, 'warning');
                }
                else if (type === 'reaction') {
                    triggerReaction(payload.emoji);
                    broadcast(payload, 'reaction');
                }
            };

            const handleGuestData = (data) => {
                const { type, payload } = data;
                if (type === 'users_update') setUsersInRoom(payload);
                else if (type === 'chat_message') setChatMessages(prev => [...prev, payload]);
                else if (type === 'log_entry') addLog(payload.text, payload.type, payload.id);
                else if (type === 'sync_log') setActivityLog(payload);
                else if (type === 'mic_status') updateUserStatus(payload.userId, { isMuted: payload.isMuted });
                else if (type === 'hand_update') updateUserStatus(payload.userId, { handRaised: payload.handRaised });
                else if (type === 'speaker_approved') becomeSpeaker();
                else if (type === 'speaker_revoked') stopSpeaking();
                else if (type === 'reaction') triggerReaction(payload.emoji);
                else if (type === 'role_update') updateUserStatus(payload.userId, { isSpeaker: payload.isSpeaker, handRaised: false });
                else if (type === 'kicked') { cleanup(); alert("Has sido expulsado."); }
            };

            const handleDisconnect = (peerId) => {
                let userName = "Usuario";
                setUsersInRoom(prev => {
                    const user = prev.find(u => u.id === peerId);
                    if (user) userName = user.name;
                    const updated = prev.filter(u => u.id !== peerId);
                    // Solo el Host hace broadcast cuando detecta desconexión
                    if (isHost) broadcast(updated, 'users_update');
                    return updated;
                });

                setSpeakers(prev => prev.filter(id => id !== peerId));
                
                // Limpiar llamadas
                if (callsRef.current[peerId]) {
                    callsRef.current[peerId].close();
                    delete callsRef.current[peerId];
                }
                
                // Eliminar elemento de audio DOM
                const audioEl = document.getElementById(`audio-${peerId}`);
                if (audioEl) audioEl.remove();

                if (isHost) addLogAndBroadcast(`${userName} salió`, 'danger');
            };

            const approveSpeaker = (userId) => {
                if (maxSlots > 0 && speakers.length >= maxSlots) return alert("Sala llena");
                setSpeakers(prev => [...prev, userId]);
                updateUserStatus(userId, { isSpeaker: true, handRaised: false, isMuted: true }); // Empieza muteado por seguridad
                broadcast({ userId, isSpeaker: true }, 'role_update');
                
                const user = usersInRoom.find(u => u.id === userId);
                addLogAndBroadcast(`${user?.name} subió al escenario`, 'success');

                const conn = connectionsRef.current.find(c => c.peer === userId);
                if (conn) conn.send({ type: 'speaker_approved' });
            };

            const becomeSpeaker = () => {
                // Al volverse speaker, NO activamos micro automáticamente.
                // El usuario debe elegir qué fuente activar.
                addLog("Ahora eres Speaker. Activa tu micrófono.", "success");
            };

            const stopSpeaking = () => {
                updateAudioSource('none');
                setSpeakers(prev => prev.filter(id => id !== myPeerId));
                // Cerrar llamadas salientes
                Object.values(callsRef.current).forEach(call => call.close());
                callsRef.current = {};
            };

            const cleanup = () => {
                updateAudioSource('none');
                if (peerRef.current) peerRef.current.destroy();
                setStep('lobby');
                setUsersInRoom([]);
                setChatMessages([]);
                setActivityLog([]);
                setIsHost(false);
            };
            
            const triggerReaction = (emoji) => {
                const id = Date.now() + Math.random();
                setReactions(prev => [...prev, { id, emoji, left: (Math.random() * 80 + 10) + '%' }]);
                setTimeout(() => setReactions(prev => prev.filter(r => r.id !== id)), 2000);
            };
            
            const addAudioElement = (id, stream) => {
                // Solo añadir si no existe
                if(document.getElementById(`audio-${id}`)) return;
                
                const audio = document.createElement('audio');
                audio.id = `audio-${id}`;
                audio.srcObject = stream;
                audio.autoplay = true;
                // Importante: No mutear el elemento de audio si soy yo el que habla
                // pero si es mi propio stream (loopback), sí debería estar muteado para mí.
                // PeerJS remote stream: No es mi propio stream.
                document.body.appendChild(audio);
                audio.play().catch(e => console.log("Autoplay policy:", e));
            };

            // --- RENDER ---
            if (step === 'login') return (
                <div className="min-h-screen flex items-center justify-center p-4 bg-slate-900">
                    <div className="glass-panel p-8 rounded-2xl w-full max-w-md text-center shadow-2xl">
                        <Icons.Radio size={48} className="text-blue-500 mx-auto mb-4" />
                        <h1 className="text-3xl font-bold mb-2">ZOZ</h1>
                        <Input value={username} onChange={setUsername} placeholder="Elige un nombre" icon={Icons.Users} />
                        <Button onClick={() => setStep('lobby')} disabled={!username.trim()} className="w-full mt-4">Entrar</Button>
                    </div>
                </div>
            );

            if (step === 'lobby') return (
                <div className="min-h-screen p-4 md:p-8 max-w-5xl mx-auto">
                    <header className="flex justify-between items-center mb-10">
                        <div className="flex items-center gap-2 text-xl font-bold"><Icons.Radio className="text-blue-500" /> ZOZ</div>
                        <div className="flex items-center gap-4">
                            <span className="text-slate-400">{username}</span>
                            <Button variant="ghost" onClick={() => setStep('login')}><Icons.LogOut size={18} /></Button>
                        </div>
                    </header>
                    <div className="grid md:grid-cols-2 gap-8">
                        <div className="glass-panel p-6 rounded-2xl border-t-4 border-blue-500">
                            <h3 className="text-2xl font-bold mb-4">Crear Sala</h3>
                            <Input value={roomName} onChange={setRoomName} placeholder="Nombre de la Sala" icon={Icons.Radio} />
                            <div className="mt-4">
                                <label className="text-xs text-slate-400 mb-2 block font-bold">Huecos para hablantes</label>
                                <div className="flex gap-2">
                                    {[1, 2, 3, 5, 8].map(num => (
                                        <button key={num} onClick={() => setMaxSlots(num)} className={`flex-1 py-2 rounded-lg border ${maxSlots === num ? 'bg-blue-600 border-blue-500 text-white' : 'border-slate-600 text-slate-400 hover:bg-slate-800'}`}>{num}</button>
                                    ))}
                                </div>
                            </div>
                            <Button onClick={createRoom} className="w-full mt-4"><Icons.Mic size={18} /> Iniciar</Button>
                        </div>
                        <div className="glass-panel p-6 rounded-2xl border-t-4 border-emerald-500">
                            <h3 className="text-2xl font-bold mb-4">Salas Activas</h3>
                            {activeRooms.length === 0 ? (
                                <div className="text-center py-8 text-slate-500 border-2 border-dashed border-slate-700 rounded-lg">No hay salas activas.</div>
                            ) : (
                                <div className="space-y-2 max-h-48 overflow-y-auto">
                                    {activeRooms.map((r, i) => (
                                        <div key={i} className="flex justify-between items-center p-3 bg-slate-800 rounded-lg hover:bg-slate-700 cursor-pointer" onClick={() => { setRoomName(r.name); joinRoom(r.name); }}>
                                            <div><div className="font-bold text-white">{r.name}</div><div className="text-xs text-slate-400">Host: {r.host} • {r.slots} slots</div></div>
                                            <Button variant="success" className="px-2 py-1 text-xs">Entrar</Button>
                                        </div>
                                    ))}
                                </div>
                            )}
                        </div>
                    </div>
                </div>
            );

            const amISpeaker = speakers.includes(myPeerId);

            return (
                <div className="h-screen flex bg-slate-900 overflow-hidden relative">
                    <div className="flex-1 flex flex-col relative">
                        <div className="absolute inset-0 pointer-events-none overflow-hidden">
                            {reactions.map(r => <div key={r.id} className="reaction-bubble bottom-24" style={{ left: r.left }}>{r.emoji}</div>)}
                        </div>

                        <header className="p-4 bg-slate-900/90 border-b border-slate-800 flex justify-between items-center z-10">
                            <div>
                                <h2 className="font-bold flex items-center gap-2"><span className="w-2 h-2 bg-red-500 rounded-full animate-pulse"></span> {roomName}</h2>
                                <p className="text-xs text-slate-500">{usersInRoom.length} usuarios • {speakers.length} hablantes</p>
                            </div>
                            <div className="flex gap-2">
                                <Button variant="ghost" className="z-50" onClick={() => setShowSidePanel(!showSidePanel)}>{showSidePanel ? <Icons.X /> : <Icons.MessageSquare />}</Button>
                                <Button variant="danger" onClick={cleanup} className="px-3 py-1 text-xs">Salir</Button>
                            </div>
                        </header>

                        <main className="flex-1 p-6 overflow-y-auto">
                            <div className="max-w-5xl mx-auto">
                                <h3 className="text-slate-500 text-xs font-bold uppercase tracking-wider mb-4">Escenario ({speakers.length}/{maxSlots})</h3>
                                <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-4 gap-4 mb-8">
                                    {usersInRoom.filter(u => u.isHost || u.isSpeaker).map(user => (
                                        <div key={user.id} className="relative aspect-square speaker-card group">
                                            <div className={`w-full h-full rounded-2xl flex flex-col items-center justify-center p-4 border-2 transition-all ${user.isHost ? 'border-blue-500 bg-blue-900/10' : 'border-emerald-500 bg-emerald-900/10'}`}>
                                                <div className="relative mb-3">
                                                    <div className="w-24 h-24 rounded-full bg-slate-700 flex items-center justify-center text-4xl overflow-hidden shadow-lg relative">
                                                        {user.isHost ? '👑' : '🎙️'}
                                                        {isHost && !user.isHost && (
                                                            <div className="absolute inset-0 bg-black/80 flex items-center justify-center gap-2 host-controls backdrop-blur-sm">
                                                                <button onClick={() => kickUser(user.id)} className="p-2 bg-red-600 rounded-full hover:bg-red-500 text-white" title="Expulsar"><Icons.UserX size={16} /></button>
                                                            </div>
                                                        )}
                                                    </div>
                                                    <div className="absolute -bottom-2 left-1/2 -translate-x-1/2 bg-slate-900 rounded-full px-3 py-1 border border-slate-700 shadow flex items-center gap-2">
                                                        <SoundWave isMuted={user.isMuted} />
                                                        {user.isMuted && <Icons.MicOff size={10} className="text-red-400"/>}
                                                    </div>
                                                </div>
                                                <span className="font-bold truncate w-full text-center text-lg">{user.name}</span>
                                                <span className="text-xs text-slate-400 font-medium uppercase">{user.isHost ? 'Anfitrión' : 'Hablante'}</span>
                                            </div>
                                        </div>
                                    ))}
                                    {Array.from({ length: Math.max(0, maxSlots - speakers.length) }).map((_, i) => (
                                        <div key={`slot-${i}`} className="aspect-square rounded-2xl border-2 border-dashed border-slate-800 flex flex-col items-center justify-center text-slate-600 opacity-50">
                                            <div className="w-20 h-20 rounded-full bg-slate-800 mb-2 flex items-center justify-center"><Icons.Mic size={24}/></div>
                                            <span className="text-xs font-bold">Libre</span>
                                        </div>
                                    ))}
                                </div>

                                <h3 className="text-slate-500 text-xs font-bold uppercase tracking-wider mb-4 border-t border-slate-800 pt-4">Audiencia</h3>
                                <div className="flex flex-wrap gap-2">
                                    {usersInRoom.filter(u => !u.isHost && !u.isSpeaker).map(user => (
                                        <div key={user.id} className="bg-slate-800 hover:bg-slate-750 rounded-full pl-2 pr-3 py-1.5 flex items-center gap-2 border border-slate-700 group transition-all">
                                            <div className="w-8 h-8 rounded-full bg-slate-600 flex items-center justify-center text-xs font-bold text-slate-300">{user.name.charAt(0)}</div>
                                            <span className="text-sm text-slate-200">{user.name}</span>
                                            {user.handRaised && <span className="animate-bounce text-lg">✋</span>}
                                            
                                            {isHost && (
                                                <div className="flex gap-1 ml-2 border-l border-slate-600 pl-2">
                                                    <button onClick={() => approveSpeaker(user.id)} className="text-emerald-500 hover:text-emerald-400 p-1 rounded hover:bg-emerald-500/10" title="Dar micro"><Icons.Mic size={14} /></button>
                                                    <button onClick={() => kickUser(user.id)} className="text-red-500 hover:text-red-400 p-1 rounded hover:bg-red-500/10" title="Expulsar"><Icons.UserX size={14} /></button>
                                                </div>
                                            )}
                                        </div>
                                    ))}
                                </div>
                            </div>
                        </main>
                        
                        {/* Menú de Audio Fijo en el fondo */}
                        {showAudioMenu && (isHost || amISpeaker) && (
                            <div className="audio-source-menu">
                                <div className="menu-title">Entrada de Audio</div>
                                <div className={`audio-source-btn ${audioSourceType === 'mic' ? 'active' : ''}`} onClick={() => updateAudioSource('mic')}>
                                    <Icons.Mic size={16}/> Solo Micrófono
                                </div>
                                <div className={`audio-source-btn ${audioSourceType === 'system' ? 'active' : ''}`} onClick={() => updateAudioSource('system')}>
                                    <div className="flex-1 flex items-center gap-2"><Icons.Monitor size={16}/> <span>Solo PC</span></div>
                                    <Icons.Lock size={12} className="text-emerald-500" title="Video Bloqueado"/>
                                </div>
                                <div className={`audio-source-btn ${audioSourceType === 'mixed' ? 'active' : ''}`} onClick={() => updateAudioSource('mixed')}>
                                    <div className="flex-1 flex items-center gap-2"><Icons.Layers size={16}/> <span>Mic + PC</span></div>
                                    <Icons.Lock size={12} className="text-emerald-500" title="Video Bloqueado"/>
                                </div>
                                <div className="h-px bg-slate-600 my-1"></div>
                                <div className={`audio-source-btn ${audioSourceType === 'none' ? 'active' : ''}`} onClick={() => updateAudioSource('none')}>
                                    <Icons.MicOff size={16}/> Silenciar Todo
                                </div>
                            </div>
                        )}

                        <footer className="p-4 bg-slate-800 border-t border-slate-700 flex justify-between items-center gap-4 relative z-50">
                            <div className="flex items-center gap-3 relative">
                                {(isHost || amISpeaker) && (
                                    <>
                                        <div className="flex rounded-lg bg-slate-700 p-1">
                                            <Button onClick={() => setShowAudioMenu(!showAudioMenu)} className={`!px-3 !py-2 hover:bg-slate-500 mr-1 rounded-md ${showAudioMenu ? 'bg-blue-600 text-white' : 'bg-slate-600'}`} title="Configurar Audio">
                                                <Icons.Settings size={18}/>
                                            </Button>
                                            <Button 
                                                onClick={() => updateAudioSource(audioSourceType === 'none' ? 'mic' : 'none')} 
                                                variant={audioSourceType === 'none' ? 'danger' : 'success'} 
                                                className="w-32 flex justify-between !py-2 rounded-md transition-all"
                                            >
                                                <span className="truncate">{audioSourceType === 'none' ? 'Silenciado' : audioSourceType === 'mic' ? 'En Vivo' : audioSourceType === 'system' ? 'PC Audio' : 'Mix Audio'}</span>
                                                {audioSourceType === 'none' ? <Icons.MicOff size={18} /> : <Icons.Radio size={18} className="animate-pulse" />}
                                            </Button>
                                        </div>
                                    </>
                                )}
                                {!isHost && !amISpeaker && (
                                    <Button onClick={() => { if(!handRaised) { connectionsRef.current[0].send({type:'request_speak'}); setHandRaised(true); }}} variant={handRaised ? 'ghost' : 'primary'}>
                                        <Icons.Hand size={20} /> {handRaised ? 'Esperando...' : 'Pedir turno'}
                                    </Button>
                                )}
                            </div>
                            <div className="flex gap-2">
                                {['❤️', '🔥', '👏', '😂'].map(emoji => (
                                    <button key={emoji} onClick={() => { triggerReaction(emoji); if(isHost) broadcast({emoji}, 'reaction'); else connectionsRef.current[0]?.send({type:'reaction', payload:{emoji}}); }} className="w-10 h-10 rounded-full bg-slate-700 hover:bg-slate-600 flex items-center justify-center text-lg transition hover:scale-110">{emoji}</button>
                                ))}
                            </div>
                        </footer>
                    </div>

                    <div className={`side-panel ${!showSidePanel ? 'closed' : ''}`}>
                        <div className="h-full flex flex-col">
                            <div className="flex border-b border-slate-700">
                                <button onClick={() => setActiveTab('chat')} className={`flex-1 py-3 text-sm font-semibold ${activeTab === 'chat' ? 'text-blue-400 border-b-2 border-blue-400' : 'text-slate-400'}`}>Chat</button>
                                <button onClick={() => setActiveTab('logs')} className={`flex-1 py-3 text-sm font-semibold ${activeTab === 'logs' ? 'text-blue-400 border-b-2 border-blue-400' : 'text-slate-400'}`}>Actividad</button>
                            </div>
                            <div className="flex-1 overflow-y-auto p-4 space-y-3 bg-slate-900/50">
                                {activeTab === 'chat' ? (
                                    chatMessages.map((m, i) => (
                                        <div key={i} className="bg-slate-800 p-2 rounded-lg border border-slate-700">
                                            <div className="flex justify-between items-baseline mb-1"><span className="font-bold text-xs text-blue-400">{m.sender}</span><span className="text-[10px] text-slate-500">{m.time}</span></div>
                                            <p className="text-sm text-slate-200">{m.text}</p>
                                        </div>
                                    ))
                                ) : (
                                    activityLog.map((log) => (
                                        <div key={log.id} className={`text-xs p-2 rounded border-l-2 ${log.type === 'danger' ? 'border-red-500 bg-red-500/10' : log.type === 'success' ? 'border-green-500 bg-green-500/10' : log.type === 'warning' ? 'border-yellow-500 bg-yellow-500/10' : 'border-blue-500 bg-blue-500/10'}`}>
                                            <span className="opacity-50 mr-2">{log.time}</span>{log.text}
                                        </div>
                                    ))
                                )}
                            </div>
                            {activeTab === 'chat' && (
                                <div className="p-3 border-t border-slate-700 bg-slate-800">
                                    <div className="flex gap-2">
                                        <input value={newMessage} onChange={e => setNewMessage(e.target.value)} onKeyDown={e => e.key === 'Enter' && sendChat()} placeholder="Escribe..." className="flex-1 bg-slate-900 border border-slate-700 rounded-lg px-3 py-2 text-sm focus:outline-none focus:border-blue-500" />
                                        <Button variant="primary" onClick={sendChat} className="!px-3"><Icons.Send size={16} /></Button>
                                    </div>
                                </div>
                            )}
                        </div>
                    </div>
                </div>
            );
        };

        const root = ReactDOM.createRoot(document.getElementById('root'));
        root.render(<App />);
    </script>
